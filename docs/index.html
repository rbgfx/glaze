require "glaze"

Glaze.shader(:pulse) do
  params { float :speed, default: 1.0, range: 0.0..4.0 }
  fragment do |point, resolution, uniforms|
    wave = point[0] / resolution[0] + uniforms.time * uniforms.speed
    [wave % 1.0, 0.2, 0.4, 1.0]
  end
end
